# Gates and utilities. They need no window; scripts/gates.sh runs them, and CTest registers them (labels fast, corpus, oracle, render).
function(wce_cli name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE ${ARGN})
endfunction()

wce_cli(xml_roundtrip wce_formats)
wce_cli(format_check wce_formats)
wce_cli(variation_check wce_formats wce_vfs)
wce_cli(variation_convert wce_formats)
wce_cli(rmv2_check wce_formats)
wce_cli(fxc_dump wce_formats d3d11 d3dcompiler dxguid)
wce_cli(vfs_probe wce_vfs)
wce_cli(foundation_check wce_foundation)
wce_cli(render_check wce_render wce_weather)
wce_cli(weather_check wce_formats)
wce_cli(roundtrip wce_formats)
wce_cli(order_test)
wce_cli(tdb_build wce_formats)
wce_cli(png_debug wce_formats)
wce_cli(scene_check wce_tile)
wce_cli(unit_tests wce_tile)
wce_cli(height_check wce_tile)

function(wce_test name label exe)
    add_test(NAME ${name} COMMAND ${exe} ${ARGN})
    set_tests_properties(${name} PROPERTIES LABELS ${label})
endfunction()
