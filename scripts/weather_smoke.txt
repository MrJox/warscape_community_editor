# Weather tool on the shared shell: environment documents, edits and undo, structural edits, save, one file at a time, the weather picker.
# Run by scripts/weather_check.sh (the expected values are filled in there; the projects folder is a throw-away one, $WCE_PROJECTS).
tool weather
frames 3
assert_tool weather
open "weather/charlemagne/med_sp_land_day_dry.environment"
frames 3
assert_documents 2
assert_modified 0
assert_env sun_elevation 26
assert_xml_elements %ELEMENTS%

# an attribute edit lights the scene on the next frame, is one undo entry, undo restores the file
edit_attr LIGHTING sun_direction_elevation 40
frames 3
assert_env sun_elevation 40
assert_modified 1
assert_xml_attr LIGHTING sun_direction_elevation 40
assert_history 1 1
edit_attr FOG fog_distance_start 0.8
assert_history 2 2
undo
assert_xml_attr FOG fog_distance_start 0.52
undo
frames 2
assert_env sun_elevation 26
assert_modified 0
redo
redo
assert_xml_attr FOG fog_distance_start 0.8
undo
undo

# structural edits: duplicate, add, move, remove each undo to the same file
xml_op duplicate CLOUD_ELEMENT
assert_xml_elements %ELEMENTS_PLUS_1%
assert_modified 1
xml_op add_child FOG extra
xml_op add_attr FOG note hello
xml_op down CLOUD_ELEMENT
undo
undo
undo
undo
assert_xml_elements %ELEMENTS%
assert_modified 0
xml_op duplicate CLOUD_ELEMENT
revert_xml
assert_xml_elements %ELEMENTS%
assert_modified 0
assert_history 0 0

# save: a loose file in the project's loose folder, modified marker gone, history cleared
edit_attr LIGHTING sun_direction_elevation 41
save
assert_saved weather/charlemagne/med_sp_land_day_dry.environment
assert_modified 0
assert_history 0 0
frames 2
assert_env sun_elevation 41
