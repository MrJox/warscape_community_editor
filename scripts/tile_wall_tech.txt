# The technology overlay draws procedural walls: their preview pieces' logic files (platforms, formation lines), and follows the pieces
# when the wall is made again; wall_tech_check finds no gaps, stacked platforms nor loose formation lines on a straight and a curved wall. %TILE% is the kit tile terrain_rugged_hills_desert/1x1/aa.
tile "%TILE%"
frames 5
add_polygon "WceWallSpline|roman_fort_straight|20,120;220,120"
frames 400
tile_wait_jobs
frames 3
tile_tech none
tile_tech on platforms formation pipes
tile_tech scope selected
frames 5
assert_tech 1 200
# Its pieces together: no stretch without a footprint, no platforms over one another, every formation line's ends on a platform.
wall_tech_check 0 0 0

# The wall made again (the ground under it raised): its new pieces drawn.
brush 15 6 0.7
paint raise 120 120 30
frames 400
tile_wait_jobs
frames 3
assert_tech 1 200

# A curved wall: every formation line end on a platform (the cut of the long ones along the curve: logic.rs a_long_firing_line_follows_the_bend).
select none
add_polygon "WceWallSpline|roman_fort_straight|30,300;130,250;230,300"
frames 400
tile_wait_jobs
frames 3
wall_tech_check 0 0 0

select none
frames 5
assert_tech 0
tile_tech none
quit
