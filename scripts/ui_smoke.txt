# The UI tool driven by script and the (scripted) mouse. %OUT% is the output folder; %OUT%/fixtures holds files the gate made (red.png, blue.png,
# x222.bin). Assertions are about what the tool does, with counts taken relative to what was opened; the few facts of the game's own files that
# are used are named where they are.
tool ui
frames 3
assert_tool ui
ui_open "ui/campaign ui/settlement_captured"
frames 30
ui_select 1
ui_mark_count
ui_assert_xml_roundtrip
ui_save_xml "%OUT%/before.xml"
assert_modified 0

# a drag on the canvas is one undo entry, and undo gives back the same XML
mouse drag 780 600 830 580 6
frames 3
assert_modified 1
assert_history 1 1
ui_save_xml "%OUT%/dragged.xml"
undo
frames 2
assert_modified 0
ui_save_xml "%OUT%/undone.xml"
redo
frames 2
assert_modified 1
undo

ui_select 1
ui_add testchild

ui_select testchild
ui_set x 5
undo 99
ui_assert_count_delta 0
assert_modified 0
ui_assert_xml_roundtrip
