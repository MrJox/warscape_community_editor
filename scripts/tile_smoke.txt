# Tile tool on the shared shell: select, move, rotate, duplicate, delete, polygons, layers, brush, undo and redo of all of it, save.
# Run by scripts/tile_check.sh (%TILE% is a copy of a kit tile, %OUT% a folder for the saved files).
tile "%TILE%"
frames 3
assert_tool tile
assert_instances 2
assert_type_count BUILDING_INSTANCE 2
assert_modified 0

# move: the instance follows the ground, one undo entry; undo and redo restore it
tile_select BUILDING_INSTANCE
assert_selected_type BUILDING_INSTANCE
assert_world_xz 457.556 478.767 0.01
tile_move 10 5
frames 2
assert_world_xz 467.556 483.767 0.01
assert_history 1 1
assert_modified 1
undo
frames 2
assert_world_xz 457.556 478.767 0.01
assert_modified 0
redo
frames 2
assert_world_xz 467.556 483.767 0.01
undo

# rotate, duplicate (selects the copy), delete it; everything undoes back to the file
rotate 90
duplicate
frames 2
assert_instances 3
assert_selected_type BUILDING_INSTANCE
delete_item
frames 2
assert_instances 2
undo
frames 2
assert_instances 3
undo 3
frames 2
assert_instances 2
assert_modified 0
assert_history 3 0

# a polygon and its points, a layer, moving the polygon into it
add_polygon "AI_HINT_POLYGON_INSTANCE|AIH_COVER_LINE|440,470;450,470;450,480;440,480"
frames 2
assert_instances 3
assert_selected_type AI_HINT_POLYGON_INSTANCE
polygon_point move 1 455 470
polygon_point insert 1 452 465
polygon_point remove 0
new_layer Test
move_to_layer Test
frames 2
assert_instances 3
undo 99
frames 2
assert_instances 2
assert_modified 0
redo 99
frames 2
assert_instances 3
undo 99

# the terrain brush: one stroke is one undo entry, undo restores the samples
mark_height centre 457 478
mark_height edge 520 478
paint raise 457 478 30
frames 2
assert_height_delta centre 457 478 0.05 1000
assert_height_delta edge 520 478 0 0
undo
frames 2
assert_height_delta centre 457 478 0 0
redo
frames 2
assert_height_delta centre 457 478 0.05 1000
undo
assert_modified 0

# an edit that is saved
tile_select BUILDING_INSTANCE
tile_move 3 -2
paint raise 440 470 10
save_to "%OUT%/saved"
assert_file_exists "%OUT%/saved/tile.ted"
assert_file_exists "%OUT%/saved/height_map_0.png"
tile_diagnostics "%OUT%/diagnostics.txt"
frames 2
capture 640x480 "%OUT%/capture.png" 5
quit
