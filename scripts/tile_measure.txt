# Tape measures (WceMeasureLine, TEd's Tape Measure) and angle finders (WceAngleFinder, TEd's angle finder): drawn on flat ground they
# measure what their points say, survive undo / redo, are saved in tile.wce and dropped from what the Assembly Kit gets. An angle finder
# keeps its three points. %TILE% is the kit tile farm_eastern/1x1/aa (flat where the line runs);
# %OUT% gets the saved tile, the export text and the kit copy (tile.wce has the line, the kit's tile.ted not).
tile "%TILE%"
frames 5
add_polygon "WceMeasureLine||100,100;150,100;150,140"
frames 2
assert_selected_type WceMeasureLine
tile_measure
assert_measure 90 90 90 0.05
undo
frames 2
redo
frames 2
tile_select WceMeasureLine
assert_measure 90 90 90 0.05

# An angle finder: a right angle at (100, 100); its third point moved makes 45 degrees, undo puts it back. A point inserted, or one too
# few, is refused.
add_polygon "WceAngleFinder||110,100;100,100;100,110"
frames 2
assert_selected_type WceAngleFinder
tile_angle
assert_angle 90 90
polygon_point move 2 110 110
frames 2
assert_angle 45 45
undo
frames 2
assert_angle 90 90
expect_fail polygon_point insert 1 105 100
expect_fail add_polygon "WceAngleFinder||0,0;10,0"

save_to "%OUT%/saved"
save_for_kit "%OUT%/kit/raw_data"
quit
