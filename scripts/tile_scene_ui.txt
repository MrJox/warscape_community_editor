# The Scene panel with the scripted mouse (Phase 1 of docs/plans/tile_ted_parity.md). Coordinates are pixels of the viewport image, the panel is on its left (x < 0); the layout
# is the default one of a 1584 x 921 window. %TILE% is osgiliath_test (root: Walls, Fort_Logic, rivers, Buildings, Barricades, fields, streets, Escalation 1..3, Gondorian Buildings).
tile "%TILE%"
frames 60
show_panel Scene
frames 5
assert_layer root child 0 Walls
assert_layer root child 10 "Gondorian Buildings"

# the eye and the cursor arrow of Buildings (open it first: its arrow at x -312)
mouse click -312 124
frames 4
mouse click -66 124
frames 4
assert_flag Buildings visible 0
assert_history 1 1
mouse click -44 124
frames 4
assert_flag Buildings active 0
assert_history 2 2
undo 2
frames 3
assert_flag Buildings visible 1
mouse click -312 124
frames 4

# drag "Gondorian Buildings" (row y 257) onto the top of "Walls" (row y 67): it goes in front of it, one undo entry
mouse drag -201 257 -201 59 8
frames 4
assert_layer root child 0 "Gondorian Buildings"
assert_layer root child 1 Walls
assert_history 1 1
undo
frames 4
assert_layer root child 0 Walls
assert_layer root child 10 "Gondorian Buildings"
assert_modified 0

# drag Barricades (y 143) into the middle of Fort_Logic (y 86): it becomes a child of Fort_Logic
mouse drag -201 143 -201 86 8
frames 4
assert_layer Fort_Logic/Barricades exists 1
assert_layer Barricades exists 0
undo
frames 4
assert_layer Barricades exists 1
assert_modified 0

# F2 on a selected layer starts renaming, Escape leaves it unchanged; Del deletes the selection (Walls has few items: no confirmation); undo brings it back
mouse click -250 67
frames 3
assert_selection_names Walls
key F2
frames 3
key Escape
frames 3
assert_layer Walls exists 1
mouse move -250 67
key Delete
frames 4
assert_layer Walls exists 0
undo
frames 4
assert_layer Walls exists 1
assert_modified 0
quit
