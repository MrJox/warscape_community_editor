# Cross-tool composition (Phase 7): a tile lit by a weather file of the packs (no weather document needed), then the same file open in the weather tool
# lighting the tile live while it is edited, one edit in each document, unsaved-changes prompts, Save All.
# Run by scripts/cross_tool_check.sh (%TILE% is a copy of a vanilla kit tile in a scratch kit, %KIT% that kit, %OUT% a folder for the captures; the projects folder, where Save writes the
# weather file, is a throw-away one: $WCE_PROJECTS).

# the weather tool starts with its sample map; opening a tile hides that stand-in scene and puts the tile tool on top
frames 3
assert_documents 1
assert_layer_visible sample_map 1
tile "%TILE%"
frames 3
assert_tool tile
assert_documents 2
assert_layer_visible sample_map 0
assert_layer_visible tile 1

# a tile without a weather gets the first dry one; the default light for the reference capture
expect_fail assert_preview none
tile_weather none
assert_preview none
frames 3
capture 640x360 "%OUT%/tile_default.png" 6

# the tile's own weather: a file of the packs, read without the weather tool
expect_fail tile_weather weather/charlemagne/no_such_file.environment
assert_preview none
tile_weather weather/charlemagne/med_sp_land_day_dry.environment
assert_preview weather/charlemagne/med_sp_land_day_dry.environment
frames 3
assert_env sun_elevation 26
tile_weather none
assert_preview none
frames 3

# a weather file opens next to it (the weather tool takes over); back in the tile tool it is a document of the same editor
open "weather/charlemagne/med_sp_land_day_dry.environment"
frames 3
assert_tool weather
assert_documents 3
assert_layer_visible sample_map 0
tool tile
frames 3
assert_tool tile
documents
expect_fail assert_env sun_elevation 26

# the same file open in the weather tool lights the tile live while it is edited
assert_preview none
tile_weather weather/charlemagne/med_sp_land_day_dry.environment
assert_preview weather/charlemagne/med_sp_land_day_dry.environment
frames 3
assert_env sun_elevation 26
capture 640x360 "%OUT%/tile_previewed.png" 6
edit_attr LIGHTING sun_direction_elevation 41
frames 3
assert_env sun_elevation 41
assert_modified 1 environment
capture 640x360 "%OUT%/tile_edited_light.png" 6
tile_weather none
assert_preview none
frames 3
capture 640x360 "%OUT%/tile_none.png" 6
tile_weather weather/charlemagne/med_sp_land_day_dry.environment
frames 3

# one edit in the tile; every document knows its own state (the active one is the active tool's: the tile)
tile_select BUILDING_INSTANCE
tile_move 5 5
frames 2
assert_modified 1
assert_history 1 1
assert_modified 1 environment
assert_history 1 1 environment

# closing and quitting ask first; a script says what to do with the changes
expect_fail close tile
expect_fail close environment
expect_fail close_all
assert_documents 3
close_prompt tile
assert_prompt close 1
frames 2
answer cancel
assert_prompt none
assert_documents 3
quit_prompt
assert_prompt quit 2
frames 2
answer cancel
assert_prompt none
expect_fail answer save

# Save All writes both documents; the files are checked against the originals by the script
save_all
assert_modified 0
assert_modified 0 environment
assert_saved weather/charlemagne/med_sp_land_day_dry.environment
# Save All writes the tile into the project (tile.wce), never into the kit
expect_fail assert_file_exists "%TILE%/tile.ted.bak"
echo kit copy untouched by Save All

# File > Export > For Assembly Kit into the scratch kit: tile.ted as TEd reads it, the original kept as .bak (the script compares them)
save_for_kit "%KIT%"
assert_file_exists "%TILE%/tile.ted.bak"

# closing the document (with a discarded edit) keeps the tile's weather: the saved file (the project's loose copy, read before the packs) lights it
tile_weather none
frames 3
capture 640x360 "%OUT%/tile_moved_default.png" 6
tile_weather weather/charlemagne/med_sp_land_day_dry.environment
frames 2
assert_env sun_elevation 41
edit_attr FOG fog_distance_start 0.8
assert_modified 1 environment
close environment discard
frames 3
assert_documents 2
assert_preview weather/charlemagne/med_sp_land_day_dry.environment
assert_env sun_elevation 41
capture 640x360 "%OUT%/tile_after_close.png" 6

# what the editor reads for the file now: the saved copy (the script compares it with the packs' file)
game_file weather/charlemagne/med_sp_land_day_dry.environment "%OUT%/saved.environment"

# Revert to game's version: the project's copy goes, the packs' file lights the tile again
open "weather/charlemagne/med_sp_land_day_dry.environment"
frames 2
revert_to_game environment
frames 3
expect_fail assert_saved weather/charlemagne/med_sp_land_day_dry.environment
tool tile
frames 3
assert_preview weather/charlemagne/med_sp_land_day_dry.environment
assert_env sun_elevation 26
close environment
frames 2
assert_documents 2

# the window's close button with an unsaved edit: the prompt, then discard (the tile on disk keeps what Save All wrote)
tile_select BUILDING_INSTANCE
tile_move 1 1
quit_prompt
assert_prompt quit 1
frames 2
answer discard
frames 3
echo not reached: the discard answer quits
