# Water (Phase 4 of docs/plans/tile_ted_parity.md, docs/re/water.md): river ribbons from the SplineRiverInstance elements, the sea / lake plane for the sets sea,
# sea_coast, lakes, river_mouth; the water can be switched off; a height stroke rebuilds landscape rivers on the new terrain. %TILE% is the kit tile
# river_ford/2x2_lb_rb_ford/aa (one river spline, no sea), %TILE2% lakes/10x10/aa (a lake), %OUT% a folder for the captures.
tile "%TILE%"
frames 6
assert_water 1
cam 0.5 0.35 160 2
frames 4
capture 640x360 "%OUT%/water_on.png" 6
tile_water off
frames 3
assert_water 0
capture 640x360 "%OUT%/water_off.png" 6
tile_water on
frames 3
assert_water 1
capture 640x360 "%OUT%/water_on2.png" 6
# a stroke on the ground keeps the river (rebuilt after the stroke, one mesh)
paint raise 128 128 3
frames 4
assert_water 1
undo
frames 3
assert_water 1
assert_modified 0
tile "%TILE2%"
frames 6
assert_water 1
cam 0.8 0.5 1500 2
frames 4
capture 640x360 "%OUT%/lake.png" 6
quit
